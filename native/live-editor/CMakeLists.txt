cmake_minimum_required(VERSION 3.20)
project(SC2EditorLiveData LANGUAGES CXX)
add_library(sc2-mcp-live-data SHARED memory.cpp)
target_compile_features(sc2-mcp-live-data PRIVATE cxx_std_17)
target_compile_definitions(sc2-mcp-live-data PRIVATE UNICODE _UNICODE)
if(DEFINED SC2_LIVE_BUILD_TAG)
 target_compile_definitions(sc2-mcp-live-data PRIVATE SC2_LIVE_BUILD_TAG="${SC2_LIVE_BUILD_TAG}")
endif()
target_link_libraries(sc2-mcp-live-data PRIVATE user32 comctl32 gdi32)
set_property(TARGET sc2-mcp-live-data PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(sc2-mcp-live-loader memory_loader.cpp)
target_compile_features(sc2-mcp-live-loader PRIVATE cxx_std_17)
target_compile_definitions(sc2-mcp-live-loader PRIVATE UNICODE _UNICODE)
target_link_libraries(sc2-mcp-live-loader PRIVATE user32 version)
set_property(TARGET sc2-mcp-live-loader PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
