cmake_minimum_required(VERSION 3.25)
project(SC2McpCasc LANGUAGES C CXX)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CASC_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(CASC_BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(CASC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CASC_UNICODE ON CACHE BOOL "" FORCE)
if(SC2_CASCLIB_SOURCE)
  add_subdirectory("${SC2_CASCLIB_SOURCE}" casclib)
else()
  include(FetchContent)
  FetchContent_Declare(CascLib GIT_REPOSITORY https://github.com/ladislav-zezula/CascLib.git GIT_TAG 38a34665624b8775bb875274b36191b21c38d97b)
  FetchContent_MakeAvailable(CascLib)
endif()
add_executable(sc2-mcp-casc casc.cpp)
target_compile_features(sc2-mcp-casc PRIVATE cxx_std_17)
target_link_libraries(sc2-mcp-casc PRIVATE casc_static)
if(WIN32)
  target_compile_definitions(sc2-mcp-casc PRIVATE UNICODE _UNICODE)
endif()
if(MSVC)
  target_compile_options(sc2-mcp-casc PRIVATE /utf-8)
endif()
