cmake_minimum_required(VERSION 3.25)
project(SC2McpArchive LANGUAGES C CXX)
set(STORM_SKIP_INSTALL ON CACHE BOOL "" FORCE)
set(STORM_USE_BUNDLED_LIBRARIES ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/StormLib)
add_executable(sc2-mcp-archive archive.cpp)
target_compile_features(sc2-mcp-archive PRIVATE cxx_std_17)
target_link_libraries(sc2-mcp-archive PRIVATE storm)
if(WIN32)
  target_compile_definitions(storm PRIVATE UNICODE _UNICODE)
  target_compile_definitions(sc2-mcp-archive PRIVATE UNICODE _UNICODE)
  if(MINGW)
    target_link_options(sc2-mcp-archive PRIVATE -municode -static)
  endif()
endif()
